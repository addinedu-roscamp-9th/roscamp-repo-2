핑키 #1 포트: 5000

libcamera-vid -t 0 \
  --width 640 --height 360 --framerate 30 \
  --codec h264 --profile baseline \
  --inline --intra 15 \
  --bitrate 650000 \
  --nopreview --rotation 180 \
  --libav-format h264 -o - | \
gst-launch-1.0 -v fdsrc do-timestamp=true ! \
  h264parse config-interval=-1 ! \
  rtph264pay config-interval=1 pt=96 mtu=1200 ! \
  udpsink host=192.168.1.8 port=5000 sync=false

로봇팔 포트: 5002

gst-launch-1.0 -v \
  v4l2src device=/dev/video0 ! \
  video/x-raw,width=640,height=480,framerate=30/1 ! \
  videoconvert ! \
  x264enc tune=zerolatency bitrate=1200 speed-preset=ultrafast key-int-max=20 ! \
  h264parse ! \
  rtph264pay config-interval=1 pt=96 ! \
  udpsink host=192.168.1.8 port=5002 sync=false
